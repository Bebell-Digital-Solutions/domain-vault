            activateUser(email) {
                // Remove from pending
                mockData.pendingUsers = mockData.pendingUsers.filter(u => u.email !== email);
                // Update in users array
                const user = mockData.users.find(u => u.email === email);
                if (user) user.status = 'active';
                
                this.renderAll();
                this.showToast(`User ${email} has been activated.`, 'success');
            },

            getStatusBadgeClass(status) {
                switch(status.toLowerCase()) {
                    case 'active':
                    case 'completed': return 'badge-success';
                    case 'pending': return 'badge-warning';
                    case 'suspended':
                    case 'rejected':
                    case 'reversed':
                    case 'failed': return 'badge-danger';
                    case 'refunded': return 'badge-info';
                    default: return 'bg-slate-500/10 text-slate-500';
                }
            },

            showToast(message, type = 'success') {
                const toast = document.getElementById('toast');
                const title = document.getElementById('toastTitle');
                const msg = document.getElementById('toastMsg');
                const icon = document.getElementById('toastIcon');

                title.textContent = type === 'success' ? 'Success' : (type === 'error' ? 'Error' : 'Information');
                msg.textContent = message;

                // Reset classes
                toast.className = 'fixed bottom-6 right-6 transform transition-all duration-300 z-[100] max-w-sm w-full';
                const inner = toast.querySelector('div');
                
                if(type === 'success') {
                    inner.className = 'bg-surface-hover border-l-4 border-emerald-500 rounded-lg shadow-2xl p-4 flex items-start gap-3';
                    icon.setAttribute('data-lucide', 'check-circle');
                    icon.className = 'w-5 h-5 text-emerald-500 shrink-0';
                } else if(type === 'error') {
                    inner.className = 'bg-surface-hover border-l-4 border-red-500 rounded-lg shadow-2xl p-4 flex items-start gap-3';
                    icon.setAttribute('data-lucide', 'alert-circle');
                    icon.className = 'w-5 h-5 text-red-500 shrink-0';
                } else {
                    inner.className = 'bg-surface-hover border-l-4 border-blue-500 rounded-lg shadow-2xl p-4 flex items-start gap-3';
                    icon.setAttribute('data-lucide', 'info');
                    icon.className = 'w-5 h-5 text-blue-500 shrink-0';
                }
                
                lucide.createIcons();
                
                // Show
                requestAnimationFrame(() => {
                    toast.classList.remove('translate-y-full', 'opacity-0');
                    toast.classList.add('translate-y-0', 'opacity-100');
                });

                // Hide after 3s
                setTimeout(() => {
                    toast.classList.remove('translate-y-0', 'opacity-100');
                    toast.classList.add('translate-y-full', 'opacity-0');
                }, 3000);
            }
        };

        // Initialize application when DOM is ready
        document.addEventListener('DOMContentLoaded', () => app.init());
    </script>


<script>
	const canvas = document.getElementById("canvas");
	const ctx = canvas.getContext("2d");

	const fontSize = 2;
	const columns = canvas.width / fontSize;
	const drops = new Array(columns).fill(1);
	function draw() {
		ctx.fillStyle = "rgba(0, 0, 0, 0.05)";
		ctx.fillRect(0, 0, canvas.width, canvas.height);
		ctx.fillStyle = "#ff5011";
		ctx.font = fontSize + "px arial";

		for (let i = 0; i < drops.length; i++) {
			const text = String.fromCharCode(Math.floor(Math.random() * 94 + 33));
			ctx.fillText(text, i * fontSize, drops[i] * fontSize);

			if (drops[i] * fontSize > canvas.height && Math.random() > 0.975) {
				drops[i] = 0;
			}

			drops[i]++;
		}
	}

	setInterval(draw, 33);
</script>
